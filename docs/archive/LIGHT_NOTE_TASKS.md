```bash
export JAVA_HOME="$(/usr/libexec/java_home -v 21)"
bash scripts/test-note-isolation.sh
(cd pnkx-admin && mvn -pl pnkx-admin -am test)
node --test pnkx-ui/tests/*.test.mjs pnkx-uniapp/tests/note-offline.test.mjs pnkx-uniapp/tests/offline-sync-contract.test.mjs
(cd pnkx-ui && npx eslint --ext .vue,.js src/views/px/workspace src/components/ResourceLinks src/components/WorkspaceSummary src/api/workspace.js src/utils/noteSession.js)
(cd pnkx-ui && npm run build)
```
